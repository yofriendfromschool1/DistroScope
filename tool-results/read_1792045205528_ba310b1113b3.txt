    73→function MatchRing({
    74→  value,
    75→  size = 96,
    76→  label,
    77→}: {
    78→  value: number;
    79→  size?: number;
    80→  label?: string;
    81→}) {
    82→  const gradId = useId();
    83→  const r = (size - 12) / 2;
    84→  const c = 2 * Math.PI * r;
    85→  return (
    86→    <div
    87→      className="relative shrink-0"
    88→      style={{ width: size, height: size }}
    89→      role="img"
    90→      aria-label={`${value} percent match${label ? ` for ${label}` : ""}`}
    91→    >
    92→      <svg width={size} height={size} className="-rotate-90">
    93→        <defs>
    94→          <linearGradient id={gradId} x1="0%" y1="0%" x2="100%" y2="100%">
    95→            <stop offset="0%" stopColor="#a78bfa" />
    96→            <stop offset="100%" stopColor="#e879f9" />
    97→          </linearGradient>
    98→        </defs>
    99→        <circle
   100→          cx={size / 2}
   101→          cy={size / 2}
   102→          r={r}
   103→          fill="none"
   104→          stroke="rgba(255,255,255,0.08)"
   105→          strokeWidth={7}
   106→        />
   107→        <motion.circle
   108→          cx={size / 2}
   109→          cy={size / 2}
   110→          r={r}
   111→          fill="none"
   112→          stroke={`url(#${gradId})`}
   113→          strokeWidth={7}
   114→          strokeLinecap="round"
   115→          strokeDasharray={c}
   116→          initial={{ strokeDashoffset: c }}
   117→          animate={{ strokeDashoffset: c - (c * value) / 100 }}
   118→          transition={{ duration: 1.3, ease: [0.22, 1, 0.36, 1], delay: 0.2 }}
   119→          style={{
   120→            filter: "drop-shadow(0 0 6px rgba(232,121,249,0.45))",
   121→          }}
   122→        />
   123→      </svg>
   124→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   125→        <span
   126→          className="font-mono font-bold leading-none"
   127→          style={{ fontSize: size / 4 }}
   128→        >
   129→          {value}
   130→          <span className="text-[0.6em]">%</span>
   131→        </span>
   132→        {label && (
   133→          <span className="mt-1 max-w-full truncate text-[9px] uppercase tracking-wider text-muted-foreground">
   134→            {label}
   135→          </span>
   136→        )}
   137→      </div>
   138→    </div>
   139→  );
   140→}

   142→export function QuizSection() {
   143→  const [phase, setPhase] = useState<Phase>("intro");
   144→  const [index, setIndex] = useState(0);
   145→  const [direction, setDirection] = useState(1);
   146→  const [answers, setAnswers] = useState<QuizAnswers>({});
   147→  const [result, setResult] = useState<MatchResponse | null>(null);
   148→  const [analyzingMsg, setAnalyzingMsg] = useState(0);
   149→  const [feedbackGiven, setFeedbackGiven] = useState<"up" | "down" | null>(null);
   150→  const [selectedDistro, setSelectedDistro] = useState<Distro | null>(null);
   151→  const [aiExplanation, setAiExplanation] = useState<string | null>(null);
   152→  const [aiLoading, setAiLoading] = useState(false);
   153→  const [aiFailed, setAiFailed] = useState(false);
   154→  const [answersOpen, setAnswersOpen] = useState(false);
   155→  const advancing = useRef(false);
   156→  const restoredRef = useRef(false);
   157→  const { toast } = useToast();
   158→
   159→  const setHighlight = useDistroScope((s) => s.setHighlight);
   160→  const compare = useDistroScope((s) => s.compare);
   161→  const toggleCompare = useDistroScope((s) => s.toggleCompare);
   162→
   163→  const question: QuizQuestion | undefined = QUIZ_QUESTIONS[index];
   164→  const matches = result?.matches ?? [];
   165→  const top = matches[0];
   166→  const selectedIds = question ? (answers[question.id] ?? []) : [];
   167→
   168→  const analyze = useCallback(
   169→    async (finalAnswers: QuizAnswers) => {
   170→      setPhase("analyzing");
   171→      setAnalyzingMsg(0);
   172→      try {
   173→        const [res] = await Promise.all([
   174→          fetch("/api/match", {
   175→            method: "POST",
   176→            headers: { "Content-Type": "application/json" },
   177→            body: JSON.stringify({ answers: finalAnswers }),
   178→          }),
   179→          new Promise((r) => setTimeout(r, 1900)),
   180→        ]);
   181→        if (!res.ok) throw new Error("match failed");
   182→        const data = (await res.json()) as MatchResponse;
   183→        setResult(data);
   184→        setFeedbackGiven(null);
   185→        setAiExplanation(null);
   186→        setAiFailed(false);
   187→        setPhase("results");
   188→        // deep-link: make this result shareable via ?q=<code>
   189→        try {
   190→          const code = encodeQuizAnswers(finalAnswers);
   191→          window.history.replaceState(
   192→            null,
   193→            "",
   194→            `${window.location.pathname}?q=${code}#finder`
   195→          );
   196→        } catch {
   197→          // non-critical
   198→        }
   199→        window.dispatchEvent(new Event("distroscope:quiz-done"));
   200→      } catch {
   201→        toast({
   202→          title: "Something went wrong",
   203→          description: "Could not score your answers. Let's try that again.",
   204→          variant: "destructive",
   205→        });
   206→        setPhase("quiz");
   207→      }
   208→    },
   209→    [toast]
   210→  );
   211→
   212→  // restore a shared result from ?q=<code> deep-link
   213→  useEffect(() => {
   214→    if (restoredRef.current) return;
   215→    const code = new URLSearchParams(window.location.search).get("q");
   216→    if (!code) return;
   217→    const restored = decodeQuizAnswers(code);
   218→    if (!restored) return;
   219→    restoredRef.current = true;
   220→    setAnswers(restored);
   221→    analyze(restored);
   222→  }, [analyze]);
   223→
   224→  const advance = useCallback(() => {
   225→    if (advancing.current) return;
   226→    advancing.current = true;
   227→    setDirection(1);
   228→    if (index < QUIZ_QUESTIONS.length - 1) {
   229→      setIndex((i) => i + 1);
   230→      setTimeout(() => (advancing.current = false), 220);
   231→    } else {
   232→      analyze(answers);
   233→      advancing.current = false;
   234→    }
   235→  }, [index, answers, analyze]);
   236→
   237→  const goBack = useCallback(() => {
   238→    if (index === 0) return;
   239→    setDirection(-1);
   240→    setIndex((i) => i - 1);
   241→  }, [index]);
   242→
   243→  // jump back to a question from the results "your answers" panel —
   244→  // continuing through to the end re-analyzes with the updated answers
   245→  const editAnswer = useCallback(
   246→    (qIndex: number) => {
   247→      setDirection(qIndex < index ? -1 : 1);
   248→      setIndex(qIndex);
   249→      setPhase("quiz");
   250→    },
   251→    [index]
   252→  );
   253→
   254→  const selectOption = useCallback(
   255→    (q: QuizQuestion, optionId: string) => {
   256→      if (q.type === "single") {
   257→        setAnswers((a) => ({ ...a, [q.id]: [optionId] }));
   258→        setTimeout(() => advance(), 240);
   259→      } else {
   260→        setAnswers((a) => {
   261→          const current = a[q.id] ?? [];
   262→          const next = current.includes(optionId)
   263→            ? current.filter((x) => x !== optionId)
   264→            : current.length >= 3
   265→              ? current
   266→              : [...current, optionId];
   267→          return { ...a, [q.id]: next };
   268→        });
   269→      }
   270→    },
   271→    [advance]
   272→  );
   273→
   274→  // rotating analyzing messages
   275→  useEffect(() => {
   276→    if (phase !== "analyzing") return;
   277→    const t = setInterval(
   278→      () =>
   279→        setAnalyzingMsg(
   280→          (m) => (m + 1) % ANALYZING_MESSAGES.length
   281→        ),
   282→      560
   283→    );
   284→    return () => clearInterval(t);
   285→  }, [phase]);
   286→
   287→  // keyboard navigation
   288→  useEffect(() => {
   289→    if (phase !== "quiz" || !question) return;
   290→    const handler = (e: KeyboardEvent) => {
   291→      if (e.target instanceof HTMLInputElement) return;
   292→      if (/^[1-9]$/.test(e.key)) {
   293→        const opt = question.options[parseInt(e.key, 10) - 1];
   294→        if (opt) {
   295→          e.preventDefault();
   296→          selectOption(question, opt.id);
   297→        }
   298→      } else if (e.key === "Enter" && question.type === "multi") {
   299→        if ((answers[question.id] ?? []).length > 0) {
   300→          e.preventDefault();
   301→          advance();
   302→        }
   303→      } else if (e.key === "ArrowLeft" && index > 0) {
   304→        e.preventDefault();
   305→        goBack();
   306→      }
   307→    };
   308→    window.addEventListener("keydown", handler);
   309→    return () => window.removeEventListener("keydown", handler);
   310→  }, [phase, question, answers, index, selectOption, advance, goBack]);
   311→
   312→  const retake = () => {
   313→    setAnswers({});
   314→    setResult(null);
   315→    setIndex(0);
   316→    setDirection(1);
   317→    setAiExplanation(null);
   318→    setAiFailed(false);
   319→    setPhase("intro");
   320→    // clear the deep-link param so a fresh quiz isn't shadowed by the old one
   321→    try {
   322→      window.history.replaceState(
   323→        null,
   324→        "",
   325→        `${window.location.pathname}#finder`
   326→      );
   327→    } catch {
   328→      // non-critical
   329→    }
   330→  };
   331→
   332→  const viewOnGraph = () => {
   333→    setHighlight(matches.map((m) => m.distro.id));
   334→    document
   335→      .getElementById("explore")
   336→      ?.scrollIntoView({ behavior: "smooth" });
   337→    toast({
   338→      title: "Matches highlighted on the map",
   339→      description: "Scroll over the bubbles glowing to compare.",
   340→    });
   341→  };
   342→
   343→  const share = async () => {
   344→    if (!top) return;
   345→    const link = buildShareUrl(answers);
   346→    const lines = [
   347→      `DistroScope says my perfect Linux is ${top.distro.name} (${top.score}%)`,
   348→      ...matches
   349→        .slice(1, 3)
   350→        .map((m) => `· ${m.distro.name} (${m.score}%)`),
   351→      `See why — or find yours: ${link}`,
   352→    ];
   353→    try {
   354→      await navigator.clipboard.writeText(lines.join("\n"));
   355→      toast({
   356→        title: "Shareable link copied",
   357→        description:
   358→          "Anyone opening it lands straight on these results — answers included.",
   359→      });
   360→    } catch {
   361→      toast({ title: "Could not copy", variant: "destructive" });
   362→    }
   363→  };
   364→
   365→  const askOracle = async () => {
   366→    if (aiLoading || aiExplanation) return;
   367→    setAiLoading(true);
   368→    setAiFailed(false);
   369→    try {
   370→      const res = await fetch("/api/explain", {
   371→        method: "POST",
   372→        headers: { "Content-Type": "application/json" },
   373→        body: JSON.stringify({ answers }),
   374→      });
   375→      if (!res.ok) throw new Error("explain failed");
   376→      const data = (await res.json()) as { explanation: string };
   377→      setAiExplanation(data.explanation);
   378→    } catch {
   379→      setAiFailed(true);
   380→      toast({
   381→        title: "The oracle is silent",
   382→        description: "AI explanation is unavailable — the match reasons still stand.",
   383→        variant: "destructive",
   384→      });
   385→    } finally {
   386→      setAiLoading(false);
   387→    }
   388→  };
   389→
   390→  const sendFeedback = async (helpful: boolean) => {
   391→    if (!top) return;
   392→    setFeedbackGiven(helpful ? "up" : "down");
   393→    try {
   394→      await fetch("/api/feedback", {
   395→        method: "POST",
   396→        headers: { "Content-Type": "application/json" },
   397→        body: JSON.stringify({ distroName: top.distro.name, helpful }),
   398→      });
   399→      window.dispatchEvent(new Event("distroscope:feedback-given"));
   400→    } catch {
   401→      // non-critical
   402→    }
   403→  };
   404→
   405→  return (
   406→    <section
   407→      id="finder"
   408→      className="relative mx-auto w-full max-w-7xl scroll-mt-24 px-4 py-16 sm:px-6 sm:py-20"
   409→    >
   410→      <header className="mx-auto max-w-2xl text-center">
   411→        <div className="glass-soft inline-flex items-center gap-2 rounded-full px-3.5 py-1.5 text-xs font-medium text-muted-foreground">
   412→          <ListChecks className="size-3.5 text-fuchsia-300" />
   413→          Distro finder
   414→        </div>
   415→        <h2 className="mt-4 text-balance text-3xl font-bold tracking-tight sm:text-4xl">
   416→          Eleven questions between you and{" "}
   417→          <span className="text-gradient">your distro</span>
   418→        </h2>
   419→        <p className="mt-3 text-pretty leading-relaxed text-muted-foreground">
   420→          Device, experience, priorities, quirks — the quiz listens to all of
   421→          it and ranks all 42 distros for you.
   422→        </p>
   423→      </header>
   424→
   425→      <div className="mx-auto mt-10 max-w-3xl">
   426→        <AnimatePresence mode="wait">
   427→          {/* ── Intro ── */}
   428→          {phase === "intro" && (
   429→            <motion.div
   430→              key="intro"
   431→              initial={{ opacity: 0, y: 24 }}
   432→              animate={{ opacity: 1, y: 0 }}
   433→              exit={{ opacity: 0, y: -16 }}
   434→              transition={{ duration: 0.45 }}
   435→              className="glass relative overflow-hidden rounded-3xl p-7 sm:p-9"
   436→            >
   437→              <div className="pointer-events-none absolute -right-20 -top-20 size-56 rounded-full bg-violet-500/15 blur-[70px]" />
   438→              <div className="pointer-events-none absolute -bottom-24 -left-16 size-56 rounded-full bg-fuchsia-500/10 blur-[70px]" />
   439→
   440→              <div className="relative flex flex-col items-center text-center">
   441→                <span className="grid size-14 place-items-center rounded-2xl bg-gradient-to-br from-violet-500 via-fuchsia-500 to-amber-400 shadow-[0_0_36px_rgb(192_132_252/0.4)]">
   442→                  <Telescope className="size-7 text-white" />
   443→                </span>
   444→                <h3 className="mt-5 text-2xl font-bold tracking-tight">
   445→                  Ready to meet your distro?
   446→                </h3>
   447→                <p className="mt-2 max-w-md text-sm leading-relaxed text-muted-foreground">
   448→                  Eleven questions, roughly two minutes, zero wrong answers.
   449→                  Your answers stay on this machine — we only count the
   450→                  result, anonymously.
   451→                </p>
   452→
   453→                <div className="mt-7 grid w-full gap-3 sm:grid-cols-3">
   454→                  {[
   455→                    {
   456→                      icon: MousePointerClick,
   457→                      title: "Answer honestly",
   458→                      text: "Device, habits, quirks — details matter",
   459→                    },
   460→                    {
   461→                      icon: Sparkles,
   462→                      title: "We score all 42",
   463→                      text: "17 weighted traits per distro",
   464→                    },
   465→                    {
   466→                      icon: Trophy,
   467→                      title: "Meet your match",
   468→                      text: "Ranked results with real reasons",
   469→                    },
   470→                  ].map((step, i) => (
   471→                    <div
   472→                      key={step.title}
   473→                      className="glass-soft rounded-2xl p-4 text-left"
   474→                    >
   475→                      <div className="flex items-center gap-2">
   476→                        <step.icon className="size-4 text-fuchsia-300" />
   477→                        <p className="text-[13px] font-semibold">
   478→                          {i + 1}. {step.title}
   479→                        </p>
   480→                      </div>
   481→                      <p className="mt-1.5 text-xs leading-relaxed text-muted-foreground">
   482→                        {step.text}
   483→                      </p>
   484→                    </div>
   485→                  ))}
   486→                </div>
   487→
   488→                <Button
   489→                  onClick={() => setPhase("quiz")}
   490→                  className="mt-8 h-12 rounded-xl bg-gradient-to-r from-violet-500 to-fuchsia-500 px-7 text-[15px] font-medium shadow-[0_10px_36px_rgb(192_132_252/0.4)] hover:brightness-110 hover:shadow-[0_10px_44px_rgb(192_132_252/0.6)]"
   491→                >
   492→                  <Sparkles className="size-4" />
   493→                  Start the quiz
   494→                </Button>
   495→                <p className="mt-3 font-mono text-[11px] text-muted-foreground/70">
   496→                  tip: keys 1–9 pick options · ← goes back
   497→                </p>
   498→              </div>
   499→            </motion.div>
   500→          )}
   501→
   502→          {/* ── Quiz ── */}
   503→          {phase === "quiz" && question && (
   504→            <motion.div
   505→              key="quiz"
   506→              initial={{ opacity: 0 }}
   507→              animate={{ opacity: 1 }}
   508→              exit={{ opacity: 0 }}
   509→              className="glass relative overflow-hidden rounded-3xl p-5 sm:p-8"
   510→            >
   511→              {/* progress */}
   512→              <div className="flex items-center justify-between gap-4">
   513→                <button
   514→                  type="button"
   515→                  onClick={goBack}
   516→                  disabled={index === 0}
   517→                  className={cn(
   518→                    "glass-soft flex items-center gap-1 rounded-lg px-2.5 py-1.5 text-xs font-medium text-muted-foreground transition-all",
   519→                    index === 0
   520→                      ? "pointer-events-none opacity-30"
   521→                      : "hover:bg-white/10 hover:text-foreground"
   522→                  )}
   523→                >
   524→                  <ChevronLeft className="size-3.5" />
   525→                  Back
   526→                </button>
   527→                <div className="flex flex-1 gap-1 px-2" aria-hidden>
   528→                  {QUIZ_QUESTIONS.map((q, i) => (
   529→                    <div
   530→                      key={q.id}
   531→                      className={cn(
   532→                        "h-1 flex-1 rounded-full transition-colors duration-300",
   533→                        i < index
   534→                          ? "bg-fuchsia-400/60"
   535→                          : i === index
   536→                            ? "bg-gradient-to-r from-violet-400 to-fuchsia-400"
   537→                            : "bg-white/[0.08]"
   538→                      )}
   539→                    />
   540→                  ))}
   541→                </div>
   542→                <p className="shrink-0 font-mono text-xs text-muted-foreground">
   543→                  {index + 1}/{QUIZ_QUESTIONS.length}
   544→                </p>
   545→              </div>
   546→
   547→              {/* question */}
   548→              <AnimatePresence mode="wait" custom={direction}>
   549→                <motion.div
   550→                  key={question.id}
   551→                  custom={direction}
   552→                  variants={slide}
   553→                  initial="enter"
   554→                  animate="center"
   555→                  exit="exit"
   556→                  transition={{ duration: 0.32, ease: [0.22, 1, 0.36, 1] }}
   557→                  className="mt-7"
   558→                >
   559→                  <div className="flex items-start gap-4">
   560→                    <span className="glass-soft grid size-12 shrink-0 place-items-center rounded-2xl">
   561→                      <DistroIcon
   562→                        name={question.icon}
   563→                        className="size-6 text-fuchsia-300"
   564→                      />
   565→                    </span>
   566→                    <div className="min-w-0">
   567→                      <h3 className="flex flex-wrap items-center gap-2 text-xl font-bold leading-snug tracking-tight sm:text-2xl">
   568→                        <span className="text-balance">{question.title}</span>
   569→                        {question.why && (
   570→                          <TooltipProvider delayDuration={120}>
   571→                            <Tooltip>
   572→                              <TooltipTrigger asChild>
   573→                                <button
   574→                                  type="button"
   575→                                  aria-label="Why we ask this"
   576→                                  className="glass-soft group grid size-6 shrink-0 place-items-center rounded-full border-white/10 text-muted-foreground transition-all hover:border-fuchsia-300/40 hover:text-fuchsia-200"
   577→                                >
   578→                                  <Info className="size-3.5 transition-transform group-hover:scale-110" />
   579→                                </button>
   580→                              </TooltipTrigger>
